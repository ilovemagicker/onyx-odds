> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a mobile-first Next.js generic N-option probability tracker at `odds.onyxlab.dev` with frequency + due stats, button and paste input, and localStorage persistence.

**Architecture:** Next.js App Router (Vercel) hosts a single-page Onyx shell. Pure TypeScript modules own options, result sequence, stats, and paste parsing; a thin localStorage adapter persists state. No backend, wallet, or auth.

**Tech Stack:** Next.js 15, React 19, TypeScript, Tailwind CSS 4, Vitest, Vercel.

## Global Constraints

- Brand: Onyx Labs; domain `odds.onyxlab.dev`; working name Odds; repo `ilovemagicker/onyx-odds`.
- UI language: Traditional Chinese primary.
- Options: N in **2–20**; custom labels; ids stored as `0..N-1`.
- Prediction display: **both** hottest (frequency) and most due (longest gap); user judges.
- Input: option buttons **and** paste (comma / whitespace / newline); unknown paste tokens reject entire paste.
- Persistence: `localStorage` key `onyx-odds:v1` only.
- Disclaimer always visible: 歷史統計僅供參考，不影響真實下一局機率.
- No accounts, wallets, real money, backend API, or onyxlab.dev homepage card in v1.
- Spec: `docs/superpowers/specs/2026-10-25-onyx-odds-design.md`.

```ts
// lib/options.ts
import type { Option } from "./types";

export const MIN_OPTIONS = 2;
export const MAX_OPTIONS = 20;

const LETTERS = "ABCDEFGHIJKLMNOPQRST"; // 20 chars

function clampN(n: number): number {
  if (!Number.isFinite(n)) return MIN_OPTIONS;
  return Math.min(MAX_OPTIONS, Math.max(MIN_OPTIONS, Math.trunc(n)));
}

export function defaultOptions(n: number): Option[] {
  const count = clampN(n);
  return Array.from({ length: count }, (_, id) => ({
    id,
    label: LETTERS[id] ?? `選項${id + 1}`,
  }));
}

export function resizeOptions(
  options: Option[],
  n: number,
): { options: Option[]; resultsCleared: true } {
  const count = clampN(n);
  const next: Option[] = [];
  for (let id = 0; id < count; id++) {
    const prev = options.find((o) => o.id === id);
    next.push({
      id,
      label: prev?.label ?? LETTERS[id] ?? `選項${id + 1}`,
    });
  }
  return { options: next, resultsCleared: true };
}

export function renameOption(
  options: Option[],
  id: number,
  label: string,
): Option[] {
  const trimmed = label.trim();
  return options.map((o) => {
    if (o.id !== id) return o;
    return {
      id: o.id,
      label: trimmed.length > 0 ? trimmed : o.label || `選項${id + 1}`,
    };
  });
}
```

```ts
import type { Option } from "./types";

export type OptionStat = {
  id: number;
  label: string;
  count: number;
  frequency: number;
  fair: number;
  gap: number;
  isHot: boolean;
  isDue: boolean;
};

export type StatsSummary = {
  total: number;
  rows: OptionStat[];
  hottestIds: number[];
  dueIds: number[];
};

export function computeStats(
  options: Option[],
  results: number[],
): StatsSummary {
  const N = options.length;
  const T = results.length;
  const fair = N > 0 ? 1 / N : 0;

  const counts = new Map<number, number>();
  for (const o of options) counts.set(o.id, 0);
  for (const id of results) {
    if (counts.has(id)) counts.set(id, (counts.get(id) ?? 0) + 1);
  }

  const lastIndex = new Map<number, number>();
  results.forEach((id, idx) => {
    lastIndex.set(id, idx);
  });

  const rowsBase = options.map((o) => {
    const count = counts.get(o.id) ?? 0;
    const frequency = T > 0 ? count / T : 0;
    const last = lastIndex.get(o.id);
    const gap = last === undefined ? T : T - 1 - last;
    return {
      id: o.id,
      label: o.label,
      count,
      frequency,
      fair,
      gap,
      isHot: false,
      isDue: false,
    };
  });

  if (T === 0) {
    return { total: 0, rows: rowsBase, hottestIds: [], dueIds: [] };
  }

  const maxFreq = Math.max(...rowsBase.map((r) => r.frequency));
  const maxGap = Math.max(...rowsBase.map((r) => r.gap));
  const hottestIds = rowsBase.filter((r) => r.frequency === maxFreq).map((r) => r.id);
  const dueIds = rowsBase.filter((r) => r.gap === maxGap).map((r) => r.id);
  const hotSet = new Set(hottestIds);
  const dueSet = new Set(dueIds);
  const rows = rowsBase.map((r) => ({
    ...r,
    isHot: hotSet.has(r.id),
    isDue: dueSet.has(r.id),
  }));

  return { total: T, rows, hottestIds, dueIds };
}
```

```ts
import type { Option } from "./types";

export type PasteResult =
  | { ok: true; ids: number[] }
  | { ok: false; error: string };

export function parsePaste(text: string, options: Option[]): PasteResult {
  const trimmed = text.trim();
  if (!trimmed) return { ok: true, ids: [] };

  const tokens = trimmed.split(/[\s,]+/).filter(Boolean);
  const byLabel = new Map(options.map((o) => [o.label, o.id]));
  const ids: number[] = [];

  for (const token of tokens) {
    if (byLabel.has(token)) {
      ids.push(byLabel.get(token)!);
      continue;
    }
    const asNum = Number(token);
    if (
      Number.isInteger(asNum) &&
      asNum >= 1 &&
      asNum <= options.length
    ) {
      ids.push(asNum - 1);
      continue;
    }
    return { ok: false, error: `無法辨識：${token}` };
  }

  return { ok: true, ids };
}
```

```ts
// lib/storage.ts
import { defaultOptions } from "./options";
import type { OddsState, Option } from "./types";

export const STORAGE_KEY = "onyx-odds:v1";

function fallback(): OddsState {
  return { options: defaultOptions(2), results: [], updatedAt: Date.now() };
}

function isOption(x: unknown): x is Option {
  return (
    typeof x === "object" &&
    x !== null &&
    typeof (x as Option).id === "number" &&
    typeof (x as Option).label === "string"
  );
}

export function loadState(storage?: Storage | null): OddsState {
  const store =
    storage ??
    (typeof globalThis !== "undefined"
      ? (globalThis as { localStorage?: Storage }).localStorage
      : undefined) ??
    null;
  if (!store) return fallback();
  try {
    const raw = store.getItem(STORAGE_KEY);
    if (!raw) return fallback();
    const parsed = JSON.parse(raw) as Partial<OddsState>;
    if (!Array.isArray(parsed.options) || !parsed.options.every(isOption)) {
      return fallback();
    }
    if (!Array.isArray(parsed.results) || !parsed.results.every((n) => typeof n === "number")) {
      return fallback();
    }
    return {
      options: parsed.options,
      results: parsed.results,
      updatedAt: typeof parsed.updatedAt === "number" ? parsed.updatedAt : Date.now(),
    };
  } catch {
    return fallback();
  }
}

export function saveState(state: OddsState, storage?: Storage | null): void {
  const store =
    storage ??
    (typeof globalThis !== "undefined"
      ? (globalThis as { localStorage?: Storage }).localStorage
      : undefined) ??
    null;
  if (!store) return;
  store.setItem(STORAGE_KEY, JSON.stringify(state));
}
```
